package store

import (
	"context"
	"path/filepath"
	"testing"
	"time"
)

// G1-3c (Messung): Wie lange hält audit rotate die Schreibsperre der
// Hauptdatenbank (WriteAuditArchive: BEGIN IMMEDIATE mit ATTACH;
// CommitAuditRotation: DELETE)? Ein zweiter Schreiber mit busy_timeout 1 s
// versucht in der Zeit einen Audit-Eintrag.
func TestG1RotateWriteLockDuration(t *testing.T) {
	if testing.Short() {
		t.Skip("measurement")
	}
	ctx := context.Background()
	dir := t.TempDir()
	st, err := Open(filepath.Join(dir, "pk.db"))
	if err != nil {
		t.Fatal(err)
	}
	defer st.Close()
	if err := st.Migrate(ctx); err != nil {
		t.Fatal(err)
	}
	const n = 400000
	if err := st.ExecForTest(`WITH RECURSIVE c(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM c WHERE i < ?)
		INSERT INTO audit_events(ts, actor_type, api_key_id, scope, operation, params_redacted, targets, result, duration_ms)
		SELECT 1700000000000 + i*60000, 'agent', 'key:abcdef', 'mail:read', 'messages.search', '{"fields":["query","limit"]}', '["m:123456"]', 'ok', 12 FROM c`, n); err != nil {
		t.Fatal(err)
	}
	ids := make([]int64, n)
	for i := range ids {
		ids[i] = int64(i + 1)
	}
	other, err := OpenWith(filepath.Join(dir, "pk.db"), Options{BusyTimeout: time.Second})
	if err != nil {
		t.Fatal(err)
	}
	defer other.Close()
	probe := func(stop <-chan struct{}, res chan<- int) {
		fails := 0
		for {
			select {
			case <-stop:
				res <- fails
				return
			default:
			}
			if err := other.ExecForTest(`INSERT INTO meta(key, value) VALUES ('g1probe', '1') ON CONFLICT(key) DO UPDATE SET value = excluded.value`); err != nil {
				fails++
			}
			time.Sleep(50 * time.Millisecond)
		}
	}
	measure := func(name string, f func() error) {
		stop, res := make(chan struct{}), make(chan int)
		go probe(stop, res)
		t0 := time.Now()
		if err := f(); err != nil {
			t.Fatal(err)
		}
		d := time.Since(t0)
		close(stop)
		t.Logf("%s: %v for %d rows, writer with busy_timeout 1s failed %d times", name, d, n, <-res)
	}
	path := st.AuditArchivePath(AuditArchiveName("2023-01-01", "2023-12-31"))
	var sum string
	measure("WriteAuditArchive", func() error {
		var err error
		sum, err = st.WriteAuditArchive(ctx, path, map[string]string{"program": "x"}, ids)
		return err
	})
	measure("CommitAuditRotation", func() error {
		return st.CommitAuditRotation(ctx, AuditArchive{File: filepath.Base(path), FromDay: "2023-01-01", ToDay: "2023-12-31",
			FirstEventID: 1, LastEventID: n, Events: n, SHA256: sum, ChainPrev: "x", ChainHead: "y", CreatedAt: time.Now()}, ids)
	})
}
