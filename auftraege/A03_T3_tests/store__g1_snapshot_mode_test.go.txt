//go:build !windows

package store

import (
	"context"
	"os"
	"path/filepath"
	"sync/atomic"
	"syscall"
	"testing"
	"time"
)

// G1-8: Der Schnappschuss der Sicherung (SnapshotTo) muss von Anfang an
// 0600 sein, nicht erst nach dem chmod (wie der Migrations-Snapshot seit T3).
func TestG1SnapshotToNeverReadableByOthers(t *testing.T) {
	old := syscall.Umask(0o022)
	defer syscall.Umask(old)
	ctx := context.Background()
	dir := t.TempDir()
	st, err := Open(filepath.Join(dir, "pk.db"))
	if err != nil {
		t.Fatal(err)
	}
	defer st.Close()
	if err := st.Migrate(ctx); err != nil {
		t.Fatal(err)
	}
	if err := st.ExecForTest(`WITH RECURSIVE c(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM c WHERE i < 200000)
		INSERT INTO audit_events(ts, actor_type, operation, params_redacted, result) SELECT i, 'system', 'sync.run', '{"x":"` +
		`aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"}', 'ok' FROM c`); err != nil {
		t.Fatal(err)
	}
	dest := filepath.Join(dir, "snap.db")
	var seen atomic.Uint32
	stop := make(chan struct{})
	done := make(chan struct{})
	go func() {
		defer close(done)
		for {
			select {
			case <-stop:
				return
			default:
			}
			if fi, err := os.Stat(dest); err == nil {
				seen.Store(seen.Load() | uint32(fi.Mode().Perm()&0o077))
			}
			time.Sleep(50 * time.Microsecond)
		}
	}()
	err = st.SnapshotTo(ctx, dest)
	close(stop)
	<-done
	if err != nil {
		t.Fatal(err)
	}
	if m := seen.Load(); m != 0 {
		t.Fatalf("snapshot was readable by group/others while it was written (bits %o)", m)
	}
}
