package retention_test

// G2-1 (Nachprüfung): blobs.gc darf keine Datei löschen, die nach der
// Altersprüfung von blob.Put erneut abgelegt wurde (Existed=true, frisches
// Datum) – etwa für einen Dateianhang eines neuen Entwurfs (drafts.create im
// MCP-Prozess, ohne Kontosperre). Entwurfsanhänge haben keine Zeile in
// blobs, also keine SQL-Nachprüfung vor Remove.
//
// Das Zeitfenster zwischen ModTime und Remove wird deterministisch über
// eine SQL-Funktion in einem Trigger auf DELETE FROM blobs geöffnet
// (DeleteUnreferencedBlobRows läuft genau dazwischen).

import (
	"bytes"
	"context"
	"database/sql/driver"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/KNUFFIXX/postkauz/internal/blob"
	"github.com/KNUFFIXX/postkauz/internal/retention"
	"github.com/KNUFFIXX/postkauz/internal/store"
	"modernc.org/sqlite"
)

var g2Hook func()

func init() {
	sqlite.MustRegisterScalarFunction("pkv_g2_hook", 0, func(*sqlite.FunctionContext, []driver.Value) (driver.Value, error) {
		if g2Hook != nil {
			h := g2Hook
			g2Hook = nil
			h()
		}
		return int64(0), nil
	})
}

func TestG2GCKeepsBlobRePutDuringRun(t *testing.T) {
	ctx := context.Background()
	dir := t.TempDir()
	st, err := store.Open(filepath.Join(dir, "pk.db"))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = st.Close() })
	if err := st.Migrate(ctx); err != nil {
		t.Fatal(err)
	}
	bs, err := blob.Open(filepath.Join(dir, "blobs"), blob.Options{})
	if err != nil {
		t.Fatal(err)
	}
	content := []byte("rechnung.pdf, früher schon einmal angehängt")
	info, err := bs.Put(bytes.NewReader(content))
	if err != nil {
		t.Fatal(err)
	}
	p, _, err := bs.Path(info.SHA256)
	if err != nil {
		t.Fatal(err)
	}
	old := time.Now().Add(-90 * 24 * time.Hour)
	if err := os.Chtimes(p, old, old); err != nil {
		t.Fatal(err)
	}
	// Eine alte, unverwiesene Zeile ohne Datei, damit DeleteUnreferencedBlobRows läuft.
	other := "aa" + string(bytes.Repeat([]byte("0"), 62))
	if err := st.ExecForTest(`INSERT INTO blobs(sha256,size,compressed,created_at) VALUES(?,1,0,?)`, other, old.UnixMilli()); err != nil {
		t.Fatal(err)
	}
	if err := st.ExecForTest(`CREATE TRIGGER g2_hook BEFORE DELETE ON blobs BEGIN SELECT pkv_g2_hook(); END`); err != nil {
		t.Fatal(err)
	}
	var reput blob.Info
	g2Hook = func() {
		// drafts.create: putFiles → Put (Existed, frisches Datum); danach entstünde der Entwurf.
		reput, err = bs.Put(bytes.NewReader(content))
	}
	svc := &retention.Service{Store: st, Blobs: bs}
	rep, gcErr := svc.GC(ctx, 24*time.Hour, false)
	if gcErr != nil {
		t.Fatal(gcErr)
	}
	if err != nil || !reput.Existed {
		t.Fatalf("hook did not re-put: existed=%v err=%v", reput.Existed, err)
	}
	if !bs.Has(info.SHA256) {
		t.Fatalf("blobs.gc removed a blob that blob.Put reported as stored (Existed=true) during the run; report %+v", rep)
	}
}
