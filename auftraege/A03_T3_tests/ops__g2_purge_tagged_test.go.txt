package ops

// G2-5 (Nachprüfung, Vorschlag zur Entscheidung des Betreibers): Lokal
// getaggte Nachrichten sind bewusst markiert (E5: Tags überleben das Löschen
// auf dem Server). Der Trockenlauf von retention.purge soll sie mindestens
// zählen; nach Entscheid „behalten" bleiben sie stehen wie Entwurfsbezüge.
// Der Test prüft die strengere Variante (behalten + kept_tagged).

import (
	"context"
	"encoding/json"
	"testing"
	"time"

	"github.com/KNUFFIXX/postkauz/internal/store"
)

func TestG2PurgeKeepsAndCountsTaggedMessages(t *testing.T) {
	e := newRetEnv(t)
	ctx := context.Background()
	tagged := e.mirror(t, 1, "getaggt")
	if _, err := e.st.ChangeTags(ctx, store.TagChange{AccountID: e.acct, MessageIDs: []int64{tagged}, Add: []string{"rechnung"},
		SetBy: "cli", Now: time.Now()}); err != nil {
		t.Fatal(err)
	}
	e.goneFromServer(t, tagged, 1, 100*24*time.Hour)
	out, err := e.run(t, "retention.purge", `{"older_than":"30d"}`, true)
	if err != nil {
		t.Fatal(err)
	}
	b, _ := json.Marshal(out)
	var m map[string]any
	_ = json.Unmarshal(b, &m)
	if _, ok := m["kept_tagged"]; !ok {
		t.Errorf("dry run does not report tagged messages (no kept_tagged): %s", b)
	}
	if _, err := e.run(t, "retention.purge", `{"older_than":"30d"}`, false); err != nil {
		t.Fatal(err)
	}
	if e.count(t, `SELECT count(*) FROM messages WHERE id = ?`, tagged) != 1 {
		t.Errorf("purge deleted a message carrying the local tag 'rechnung'")
	}
}
