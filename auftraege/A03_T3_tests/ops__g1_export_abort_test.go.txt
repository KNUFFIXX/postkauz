package ops

import (
	"context"
	"encoding/json"
	"errors"
	"testing"

	"github.com/KNUFFIXX/postkauz/internal/store"
)

// G1-1: Ein Export, der zwischen Tagesdatei und Nachweis abbricht (Strg+C,
// SQLITE_BUSY > busy_timeout), darf folgende Exporte nicht dauerhaft sperren
// und kein file_changed in verify hinterlassen.
type g1FailSaveRepo struct {
	*store.SQLite
	fail bool
}

func (r *g1FailSaveRepo) SaveAuditExport(ctx context.Context, e store.AuditExport) error {
	if r.fail {
		return errors.New("simulated abort between file and record")
	}
	return r.SQLite.SaveAuditExport(ctx, e)
}

func g1Export(t *testing.T, repo AuditRepo, dir string) error {
	t.Helper()
	op := AuditExport(repo, dir)
	_, err := op.Handler(context.Background(), Call{Input: json.RawMessage(`{}`)})
	return err
}

func g1Verify(t *testing.T, repo AuditRepo) AuditVerifyOutput {
	t.Helper()
	op := AuditVerify(repo)
	res, err := op.Handler(context.Background(), Call{Input: json.RawMessage(`{}`)})
	if err != nil {
		t.Fatal(err)
	}
	return res.(AuditVerifyOutput)
}

func TestG1AuditExportAbortBetweenFileAndRecord(t *testing.T) {
	for _, tc := range []struct {
		name        string
		firstExport bool // Tag hat schon einen Nachweis (Export lief heute schon)
	}{{"recorded-day", true}, {"new-day", false}} {
		t.Run(tc.name, func(t *testing.T) {
			e := newAdminEnv(t)
			repo := &g1FailSaveRepo{SQLite: e.st}
			e.run(t, adminKey, "keys.list", map[string]any{})
			if tc.firstExport {
				if err := g1Export(t, repo, e.dir); err != nil {
					t.Fatal(err)
				}
			}
			e.run(t, adminKey, "keys.list", map[string]any{}) // neuer Eintrag heute
			repo.fail = true
			if err := g1Export(t, repo, e.dir); err == nil {
				t.Fatal("simulated abort did not happen")
			}
			repo.fail = false
			if v := g1Verify(t, repo); v.Status != "ok" {
				t.Errorf("verify right after the aborted export = %+v", v)
			}
			// Wie im Betrieb: Der abgebrochene Aufruf schreibt seinen
			// Abschluss-Eintrag (error) in den offenen Tag, der Inhalt ändert
			// sich also vor dem nächsten Export.
			e.run(t, adminKey, "keys.list", map[string]any{})
			if err := g1Export(t, repo, e.dir); err != nil {
				t.Fatalf("export after an aborted export = %v", err)
			}
			if v := g1Verify(t, repo); v.Status != "ok" {
				t.Fatalf("verify after recovery = %+v", v)
			}
		})
	}
}
