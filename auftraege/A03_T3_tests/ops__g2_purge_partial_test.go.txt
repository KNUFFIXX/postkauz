package ops

// G2-2 (Nachprüfung): Bricht retention.purge nach schon festgeschriebenen
// 500er-Paketen ab (Fehler, Strg+C), müssen die bis dahin gelöschten
// Nachrichten als Zähler im Audit-Abschluss stehen (E8: Zähler im Audit).

import (
	"context"
	"strings"
	"testing"

	"github.com/KNUFFIXX/postkauz/internal/store"
)

func TestG2PurgeAbortKeepsPartialCountsInAudit(t *testing.T) {
	e := newRetEnv(t)
	ctx := context.Background()
	if err := e.st.ExecForTest(`WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i < 600)
		INSERT INTO messages(account_id, subject, deleted_remote_at, created_at) SELECT ?, 'Weg', 1, 1 FROM n`, e.acct); err != nil {
		t.Fatal(err)
	}
	// Das zweite Paket scheitert (Ersatz für einen Fehler oder Strg+C mitten im Lauf).
	if err := e.st.ExecForTest(`CREATE TRIGGER g2_fail BEFORE DELETE ON messages
		WHEN old.id = (SELECT max(id) FROM messages) BEGIN SELECT RAISE(ABORT, 'g2 boom'); END`); err != nil {
		t.Fatal(err)
	}
	_, err := e.run(t, "retention.purge", `{"older_than":"1d"}`, false)
	if err == nil {
		t.Fatal("purge should fail in the second batch")
	}
	if left := e.count(t, `SELECT count(*) FROM messages`); left != 100 {
		t.Fatalf("messages left = %d, want 100 (first batch of 500 committed)", left)
	}
	evs, _ := e.st.AuditQuery(ctx, store.AuditFilter{Operation: "retention.purge"})
	last := evs[len(evs)-1]
	t.Logf("audit end: result=%s reason=%s params=%s", last.Result, last.Reason, last.ParamsRedacted)
	if !strings.Contains(string(last.ParamsRedacted), `"purged":500`) {
		t.Fatalf("audit end entry lacks the partial count purged=500: %s", last.ParamsRedacted)
	}
}
