package httpmcp

import (
	"net/http"
	"testing"
	"time"
)

// N8: Ist TestT5NJWKSRefetchPerBadToken scharf? Ursprungstest (ohne Uhr)
// und Variante mit Uhr, die vor jedem Token 11 s weiterläuft.
func TestZZN8RefetchOriginalShape(t *testing.T) {
	o := newOAuthEnv(t)
	if code, _, body := o.post(t, o.token(t, "user-reader", nil)); code != http.StatusOK {
		t.Fatalf("valid: %d %s", code, body)
	}
	base := o.as.jwksHits.Load()
	other := genKey(t)
	for i := 0; i < 5; i++ {
		c := o.claims("user-reader", func(c map[string]any) {
			c["iss"] = "https://evil.example.org"
			c["exp"] = time.Now().Add(-time.Hour).Unix()
		})
		_, _, _ = o.post(t, o.as.sign(t, other, "at+jwt", c))
	}
	t.Logf("original shape: %d JWKS fetches", o.as.jwksHits.Load()-base)
}

		_, _, _ = o.post(t, o.as.sign(t, other, "at+jwt", c))
	}

}
