package httpmcp

import (
	"context"
	"os"
	"strings"
	"testing"
	"time"
)

// F11 (Prüfung): Discovery mit dem Client der Produktion (HTTPClient nil →
// SDK-Discovery-Transport mit SSRF-Sperre) gegen einen AS, dessen Name auf
// eine Tailnet-Adresse zeigt (ADR-010 Punkt 8, docs/betrieb.md: issuer =
// https://postkauz.<tailnet>.ts.net:8443). Läuft nur mit ZZ_TAILNET_HOST
// (Name in /etc/hosts auf 100.x gelegt, ohne Proxy-Umgebung).
func TestZZDiscoveryToTailnetName(t *testing.T) {
	host := os.Getenv("ZZ_TAILNET_HOST")
	if host == "" {
		t.Skip("ZZ_TAILNET_HOST not set")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_, err := discoverJWKS(ctx, "https://"+host+":8443", nil)
	t.Logf("discovery error: %v", err)
	if err != nil && strings.Contains(err.Error(), "non-public") {
		t.Errorf("discovery to a tailnet AS refused by the SDK SSRF guard: %v", err)
	}
}

// F11: Aussteller als Tailnet-IP (ohne Namen) scheitert schon an der
// URL-Prüfung des SDK.
func TestZZDiscoveryToTailnetIP(t *testing.T) {
	for _, iss := range []string{"https://100.101.102.103:8443", "https://[fd7a:115c:a1e0::1]:8443"} {
		ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
		_, err := discoverJWKS(ctx, iss, nil)
		cancel()
		if err != nil && strings.Contains(err.Error(), "non-public") {
			t.Errorf("%s: refused as non-public: %v", iss, err)
		} else {
			t.Logf("%s: %v", iss, err)
		}
	}
}
