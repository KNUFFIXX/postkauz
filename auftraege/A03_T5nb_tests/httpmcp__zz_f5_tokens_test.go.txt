// F5 (Prüfung): Token-Fälle, die bisher kein Test abdeckt.
func TestZZTokenGaps(t *testing.T) {
	o := newOAuthEnv(t)
	// Cache füllen (kid k1 bekannt).
	if code, _, b := o.post(t, o.token(t, "user-reader", nil)); code != http.StatusOK {
		t.Fatalf("valid: %d %s", code, b)
	}
	b64 := func(v any) string {
		b, _ := json.Marshal(v)
		return base64.RawURLEncoding.EncodeToString(b)
	}
	claims := o.claims("user-reader", nil)
	none := b64(map[string]string{"alg": "none", "typ": "at+jwt", "kid": "k1"}) + "." + b64(claims) + "."

	pubDER, _ := x509.MarshalPKIXPublicKey(&o.as.key.PublicKey)
	hs, _ := jose.NewSigner(jose.SigningKey{Algorithm: jose.HS256, Key: pubDER}, (&jose.SignerOptions{}).WithType("at+jwt").WithHeader("kid", "k1"))
	hsTok, _ := jwt.Signed(hs).Claims(claims).Serialize()

	cases := []struct {
		name string
		tok  string
		want int
	}{
		{"alg none", none, http.StatusUnauthorized},
		{"HS256 with the public key as secret", hsTok, http.StatusUnauthorized},
		{"nbf 4 min in the future", o.token(t, "user-reader", func(c map[string]any) { c["nbf"] = time.Now().Add(4 * time.Minute).Unix() }), http.StatusUnauthorized},
		{"nbf 5 min 20 s in the future", o.token(t, "user-reader", func(c map[string]any) { c["nbf"] = time.Now().Add(5*time.Minute + 20*time.Second).Unix() }), http.StatusUnauthorized},
		{"iat one day in the future", o.token(t, "user-reader", func(c map[string]any) { c["iat"] = time.Now().Add(24 * time.Hour).Unix() }), http.StatusUnauthorized},
		{"exp in 10 years", o.token(t, "user-reader", func(c map[string]any) { c["exp"] = time.Now().AddDate(10, 0, 0).Unix() }), http.StatusOK},
	}
	for _, c := range cases {
		code, _, _ := o.post(t, c.tok)
		if code != c.want {
			t.Errorf("%s: %d, want %d", c.name, code, c.want)
		} else {
			t.Logf("%s: %d as wanted", c.name, code)
		}
	}
}

// F5: Aussteller in den Metadaten passt nicht → Discovery scheitert.
func TestZZDiscoverIssuerMismatch(t *testing.T) {
	mux := http.NewServeMux()
	srv := httptest.NewTLSServer(mux)
	defer srv.Close()
	mux.HandleFunc("/.well-known/oauth-authorization-server", func(w http.ResponseWriter, _ *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(map[string]any{"issuer": "https://evil.example.org", "jwks_uri": srv.URL + "/k",
			"authorization_endpoint": srv.URL + "/a", "token_endpoint": srv.URL + "/t", "code_challenge_methods_supported": []string{"S256"}})
	})
	if got, err := discoverJWKS(context.Background(), srv.URL, srv.Client()); err == nil {
		t.Errorf("foreign issuer in metadata accepted: %q", got)
	}
}
