package cli

import (
	"context"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/KNUFFIXX/postkauz/internal/flock"
	"github.com/KNUFFIXX/postkauz/internal/store"
)

// A11: Waehrend audit rotate laeuft (Sperre audit-rotate.lock gehalten),
// darf doctor nicht raten, die Archivdatei wegzuschieben.
func TestZZA11DoctorDuringRotate(t *testing.T) {
	dataDir := t.TempDir()
	st, err := store.Open(filepath.Join(dataDir, "postkauz.db"))
	if err != nil {
		t.Fatal(err)
	}
	defer st.Close()
	if err := st.Migrate(context.Background()); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dataDir, store.AuditArchivePrefix+"2025-01-01-2025-01-31.db"), []byte("x"), 0o600); err != nil {
		t.Fatal(err)
	}
	l, err := flock.TryLock(filepath.Join(dataDir, "audit-rotate.lock"))
	if err != nil {
		t.Fatal(err)
	}
	defer l.Release()
	var rep DoctorReport
	doctorArchiveFiles(context.Background(), st, &rep)
	for _, w := range rep.Warnings {
		if strings.Contains(w, "move them out") {
			t.Errorf("A11 bestaetigt: Rat zum Wegschieben waehrend laufender Rotation: %s", w)
		}
	}
}
