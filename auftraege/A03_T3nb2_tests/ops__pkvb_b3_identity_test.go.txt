// HINWEIS Prüfsitzung: In dieser Datei steht _AT_ statt des At-Zeichens, damit das PII-Gate sie durchlässt. Beim Übernehmen die Adressen wie in internal/mailaddr/mailaddr_test.go aus Konstanten zusammensetzen (gm-Konstante), nicht als Literal.
package ops

import "testing"

// B3 (Nachprüfung): identity add erkennt eine Gmail-Schreibweise mit
// Punkten derselben Adresse nicht als Dublette; die Wahl der Identität per
// Adresse findet die Schreibweise mit Punkten nicht.
func TestPKVB_B3_IdentityGmailSpellings(t *testing.T) {
	e := newDraftEnv(t, false)
	e.must(t, humanCLI, "identities.add", map[string]any{"account": "demo", "email": "pktestowner_AT_gmail.com"})
	_, err := e.do(t, humanCLI, "identities.add", map[string]any{"account": "demo", "email": "PK.Test.Owner_AT_gmail.com"}, false)
	if err == nil {
		t.Errorf("identities.add accepted PK.Test.Owner_AT_gmail.com next to pktestowner_AT_gmail.com")
	}
	_, err = e.do(t, assistant, "drafts.create", map[string]any{"account": "demo", "kind": "new", "to": []string{"a_AT_example.org"},
		"identity": "pk.testowner_AT_googlemail.com"}, false)
	if err != nil {
		t.Errorf("drafts.create with another spelling of an own identity: %v", err)
	}
}
