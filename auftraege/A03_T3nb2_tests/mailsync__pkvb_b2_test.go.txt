package mailsync

// B2 (Nachprüfung): Ein Fehler beim Öffnen von gc.lock, der keine belegte
// Sperre ist (hier: gc.lock ist ein Ordner), lässt blob.Put sofort
// scheitern; analyze vermerkt den Anhang dann dauerhaft als "not stored"
// (kein SHA256), die Nachricht selbst wird gespiegelt.

import (
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"github.com/KNUFFIXX/postkauz/internal/blob"
)

func TestPKVB_B2_LockOpenErrorMarksAttachmentNotStored(t *testing.T) {
	root := t.TempDir()
	bs, err := blob.Open(root, blob.Options{})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.Mkdir(filepath.Join(root, blob.LockName), 0o700); err != nil {
		t.Fatal(err)
	}
	raw := "From: a@example.org\r\nTo: b@example.org\r\nSubject: x\r\nMIME-Version: 1.0\r\n" +
		"Content-Type: multipart/mixed; boundary=B\r\n\r\n--B\r\nContent-Type: text/plain\r\n\r\nHallo\r\n" +
		"--B\r\nContent-Type: application/pdf\r\nContent-Disposition: attachment; filename=r.pdf\r\n\r\n%PDF-1.4 x\r\n--B--\r\n"
	e := &Engine{Blobs: bs}
	start := time.Now()
	_, atts, err := e.analyze(strings.NewReader(raw), time.Now())
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("analyze took %v", time.Since(start))
	for _, a := range atts {
		t.Logf("attachment %s sha=%q status=%s err=%q", a.PartID, a.SHA256, a.ExtractStatus, a.ExtractError)
		if a.SHA256 == "" {
			t.Fatalf("attachment %s stored as %q after a transient lock error (no retry)", a.PartID, a.ExtractError)
		}
	}
}
