package ops

import (
	"context"
	"os"
	"path/filepath"
	"testing"
	"time"
)

// A2: Merker vorhanden, Datei danach veraendert -> verify soll file_changed melden.
func TestZZA2MarkerThenFileChanged(t *testing.T) {
	for _, tc := range []struct {
		name     string
		recorded bool
	}{{"recorded-day", true}, {"new-day", false}} {
		t.Run(tc.name, func(t *testing.T) {
			base := t.TempDir()
			dir := filepath.Join(base, "audit-export")
			now := time.Now().UTC()
			st := t3nOpen(t, filepath.Join(base, "d", "postkauz.db"))
			repo := &t3nFailSaveRepo{SQLite: st}
			t3nEvent(t, st, now.Add(-time.Minute), "sync.run")
			if tc.recorded {
				if _, err := t3nExport(repo, dir); err != nil {
					t.Fatal(err)
				}
			}
			t3nEvent(t, st, now, "keys.list")
			repo.fail = true
			if _, err := t3nExport(repo, dir); err == nil {
				t.Fatal("no abort")
			}
			repo.fail = false
			pend, _ := st.AuditExportPending(context.Background())
			t.Logf("markers: %+v", pend)
			file := filepath.Join(dir, auditFileName(now.Format("2006-01-02")))
			b, err := os.ReadFile(file)
			if err != nil {
				t.Fatal(err)
			}
			// Datei nach dem Abbruch veraendern (eine Zeile entfernen).
			lines := 0
			for i, c := range b {
				if c == '\n' {
					lines++
					if lines == 1 {
						b = b[i+1:]
						break
					}
				}
			}
			if err := os.WriteFile(file, b, 0o600); err != nil {
				t.Fatal(err)
			}
			v := t3nVerify(t, st)
			t.Logf("verify: %s problems=%+v incomplete=%+v notExported=%v", v.Status, v.Problems, v.Incomplete, v.NotExported)
			hasChanged := false
			for _, p := range v.Problems {
				if p.Problem == AuditFileChanged || p.Problem == AuditChainBroken {
					hasChanged = true
				}
			}
			if !hasChanged {
				t.Errorf("veraenderte Datei nicht gemeldet (status %s)", v.Status)
			}
			_, xerr := t3nExport(st, dir)
			t.Logf("next export: %v", xerr)
		})
	}
}
