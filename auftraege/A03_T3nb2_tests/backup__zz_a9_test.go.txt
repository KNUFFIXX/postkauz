package backup

)

// A9: Umweg (Junction unter Windows; hier Symlink, den os.Stat genauso
// aufloest) auf einen UNTERORDNER des Datenverzeichnisses.
func TestZZA9InsideByIDLinkToSubfolder(t *testing.T) {
	base := t.TempDir()
	data := filepath.Join(base, "data")
	sub := filepath.Join(data, "blobs")
	if err := os.MkdirAll(sub, 0o700); err != nil {
		t.Fatal(err)
	}
	link := filepath.Join(base, "J")
	if err := os.Symlink(sub, link); err != nil {
		t.Skip(err)
	}
	for _, c := range []string{link, filepath.Join(link, "backup")} {
		if !insideByID(c, data) {
			t.Errorf("A9 bestaetigt: insideByID(%s -> data/blobs) = false", c)
		}
	}
}

// A9 Ende-zu-Ende mit Bind-Mount (Linux-Gegenstueck der Junction: weder
// EvalSymlinks noch Lstat sehen einen Umweg).
func TestZZA9BindMountSubfolderAsTarget(t *testing.T) {
	e := newEnv(t)
	sub := filepath.Join(e.dir, "sicherung-hier")
	if err := os.MkdirAll(sub, 0o700); err != nil {
		t.Fatal(err)
	}
	j := filepath.Join(t.TempDir(), "J")
	_ = os.Mkdir(j, 0o700)
	if out, err := zzExec("mount", "--bind", sub, j); err != nil {
		t.Skipf("mount --bind: %v %s", err, out)
	}
	defer zzExec("umount", j)
	_, _, err := e.r.prepareFolder(j)
	if err == nil {
		t.Errorf("A9 bestaetigt: Ziel %s (Umweg auf %s) angenommen", j, sub)
	}
}
