package backup

import (
	"context"
	"path/filepath"
	"testing"
)

// A8: Ein Blob, der schon VOR der Sicherung fehlte (z. B. Quarantaene des
// Virenscanners), darf nicht jede weitere Sicherung blockieren.
func TestZZA8BlobMissingBeforeBackup(t *testing.T) {
	ctx := context.Background()
	e := newEnv(t)
	to := filepath.Join(t.TempDir(), "backup")
	sum := e.sums[1]
	if err := e.st.ExecForTest(`INSERT OR IGNORE INTO blobs(sha256, size, created_at) VALUES (?, 12, 1)`, sum); err != nil {
		t.Fatal(err)
	}
	if err := e.st.ExecForTest(`UPDATE messages SET blob_sha256 = ? WHERE id = 1`, sum); err != nil {
		t.Fatal(err)
	}
	if removed, err := e.blobs.Remove(sum); err != nil || !removed {
		t.Fatalf("remove blob: %v %v", removed, err)
	}
	for i := range 2 {
		e.now = e.now.Add(60e9)
		if _, err := e.r.Create(ctx, Options{To: to}); err != nil {
			t.Errorf("A8 bestaetigt: Sicherung %d scheitert dauerhaft: %v", i+1, err)
		} else {
			t.Logf("Sicherung %d ok", i+1)
		}
	}
}
