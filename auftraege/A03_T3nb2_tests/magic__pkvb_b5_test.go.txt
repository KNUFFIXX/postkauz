package magic

import "testing"

// B5 (Nachprüfung): Endungen aus der Outlook-Sperrliste (Level 1), die
// ExecutableName nicht kennt.
func TestPKVB_B5_MoreRiskyExtensions(t *testing.T) {
	for _, ext := range []string{".ade", ".adp", ".appinstaller", ".appxbundle", ".msixbundle", ".vsto", ".ps2xml", ".psc1", ".jnlp",
		".mdb", ".accdb", ".mda", ".vb", ".wsb", ".cer", ".crt", ".der", ".ins", ".isp", ".mst", ".xnk", ".bas", ".ksh", ".csh"} {
		if !ExecutableName("rechnung" + ext) {
			t.Errorf("%s not treated as executable", ext)
		}
	}
}
