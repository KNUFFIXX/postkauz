package store_test

// B4 (Nachprüfung): Startwert id_floor_threads/attachments aus 0022 ist nur
// max(id). Ein vor 0022 beim Zusammenführen gelöschter Thread mit der
// größten ID, der im Audit als thread:<id> steht, bekommt nach 0022 seine
// ID an einen neuen Thread.

import (
	"context"
	"path/filepath"
	"testing"

	"github.com/KNUFFIXX/postkauz/internal/store"
)

func TestPKVB_B4_Floor0022CoversAuditTargets(t *testing.T) {
	ctx := context.Background()
	st, err := store.Open(filepath.Join(t.TempDir(), "pk.db"))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = st.Close() })
	if err := st.MigrateToForTest(ctx, 19); err != nil {
		t.Fatal(err)
	}
	acct, err := st.CreateAccount(ctx, store.Account{Name: "demo", Email: "demo@example.org", Host: "127.0.0.1", Port: 143,
		TLS: "none", Username: "demo@example.org", AuthType: "password", CredentialRef: "account/demo"})
	if err != nil {
		t.Fatal(err)
	}
	if err := st.ExecForTest(`INSERT INTO threads(id, account_id, subject_norm) VALUES (1, ?1, 'a'), (2, ?1, 'b');
		INSERT INTO audit_events(ts, actor_type, operation, result, targets) VALUES (1, 'agent', 'threads.get', 'ok', '["thread:2"]');
		DELETE FROM threads WHERE id = 2;`, acct); err != nil {
		t.Fatal(err)
	}
	if err := st.Migrate(ctx); err != nil {
		t.Fatal(err)
	}
	floor, err := st.QueryIntForTest(`SELECT CAST(value AS INTEGER) FROM meta WHERE key = 'id_floor_threads'`)
	if err != nil {
		t.Fatal(err)
	}
	if floor < 2 {
		t.Fatalf("id_floor_threads = %d: thread id 2 (named in the audit as thread:2) will be handed out again", floor)
	}
}
