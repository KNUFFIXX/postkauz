package ops

// B1 (Nachprüfung): Ein aus dem Spiegel übernommener Anhang (attachment_id)
// umgeht blob.Put. Laufen zwischen Einlesen und Speichern des Entwurfs
// purge (Quelle weg) und blob gc, ist der Blob weg, der Entwurf wird
// trotzdem gespeichert und verweist ins Leere.

import (
	"context"
	"io"
	"os"
	"strings"
	"testing"
	"time"

	"github.com/KNUFFIXX/postkauz/internal/blob"
	"github.com/KNUFFIXX/postkauz/internal/retention"
)

type pkvbHookBlobs struct {
	*blob.Store
	sha  string
	hook func()
}

type pkvbHookRC struct {
	io.ReadCloser
	hook func()
}

func (r *pkvbHookRC) Close() error {
	err := r.ReadCloser.Close()
	if r.hook != nil {
		h := r.hook
		r.hook = nil
		h()
	}
	return err
}

func (b *pkvbHookBlobs) Get(sha string) (io.ReadCloser, error) {
	rc, err := b.Store.Get(sha)
	if err != nil || sha != b.sha || b.hook == nil {
		return rc, err
	}
	h := b.hook
	b.hook = nil
	return &pkvbHookRC{ReadCloser: rc, hook: h}, nil
}

func TestPKVB_B1_MirroredAttachmentSurvivesPurgeAndGC(t *testing.T) {
	ctx := context.Background()
	e := newDraftEnv(t, false)
	bs, err := blob.Open(t.TempDir(), blob.Options{})
	if err != nil {
		t.Fatal(err)
	}
	content := "Rechnung 2026-117, PDF-Ersatz " + strings.Repeat("x", 100)
	info, err := bs.Put(strings.NewReader(content))
	if err != nil {
		t.Fatal(err)
	}
	p, _, _ := bs.Path(info.SHA256)
	old := time.Now().Add(-90 * 24 * time.Hour)
	if err := os.Chtimes(p, old, old); err != nil {
		t.Fatal(err)
	}
	if err := e.st.ExecForTest(`INSERT INTO blobs(sha256,size,created_at) VALUES (?,?,1)`, info.SHA256, len(content)); err != nil {
		t.Fatal(err)
	}
	if err := e.st.ExecForTest(`UPDATE attachments SET sha256 = ?, truncated = 0 WHERE id = 1`, info.SHA256); err != nil {
		t.Fatal(err)
	}
	svc := &retention.Service{Store: e.st, Blobs: bs}
	var gcRep retention.GCReport
	var gcErr error
	hb := &pkvbHookBlobs{Store: bs, sha: info.SHA256, hook: func() {
		// purge (Quelle weg) und blob gc laufen genau jetzt.
		if err := e.st.ExecForTest(`DELETE FROM attachments WHERE sha256 = ?`, info.SHA256); err != nil {
			t.Error(err)
		}
		gcRep, gcErr = svc.GC(ctx, time.Hour, false)
	}}
	e.deps.DraftBlobs = hb
	reg, err := NewDefaultRegistry(e.deps)
	if err != nil {
		t.Fatal(err)
	}
	e.x = &Executor{Registry: reg, Audit: e.st.AuditWriter()}

	out, err := e.do(t, assistant, "drafts.create", map[string]any{"account": "demo", "kind": "new", "to": []string{"a@example.org"},
		"subject": "Weiter", "attachments": []map[string]any{{"attachment_id": 1}}}, false)
	if gcErr != nil {
		t.Fatalf("gc: %v", gcErr)
	}
	if hb.hook != nil {
		t.Fatal("hook did not run")
	}
	if err != nil {
		t.Logf("drafts.create failed (acceptable outcome): %v", err)
		return
	}
	d := out.(DraftsWriteOutput)
	t.Logf("gc report: %+v; draft %d", gcRep, d.DraftID)
	refs, rerr := e.st.BlobReferences(ctx)
	t.Logf("BlobReferences names the attachment: %v (%v) -> backup create would stop with ErrIncomplete while the blob is missing", refs[info.SHA256], rerr)
	if !bs.Has(info.SHA256) {
		t.Fatalf("draft %d saved, but its attachment blob %s was removed by blob gc (report %+v)", d.DraftID, info.SHA256[:12], gcRep)
	}
}
