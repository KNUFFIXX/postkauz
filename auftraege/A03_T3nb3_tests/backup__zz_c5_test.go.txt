package backup

import (
	"context"
	"errors"
	"os"
	"path/filepath"
	"testing"
)

// C5a: Eine vollständige Sicherung eines anderen Datenverzeichnisses, deren
// Manifest fehlt (verloren, halb entpacktes Tar), gilt als Rest einer
// eigenen abgebrochenen Sicherung; der Lauf schreibt hinein und löscht
// ihren Schnappschuss.
func TestZZC5aForeignBackupWithoutManifest(t *testing.T) {
	ctx := context.Background()
	other := newEnv(t)
	to := filepath.Join(t.TempDir(), "backup")
	res, err := other.r.Create(ctx, Options{To: to})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.Remove(filepath.Join(to, ManifestName)); err != nil {
		t.Fatal(err)
	}
	e := newEnv(t)
	e.now = e.now.Add(3600e9)
	res2, err := e.r.Create(ctx, Options{To: to})
	t.Logf("second data directory into that folder: err=%v, snapshots removed=%d", err, res2.SnapshotsRemoved)
	if _, serr := os.Stat(filepath.Join(to, res.Database)); serr != nil {
		t.Errorf("the snapshot %s of the other backup is gone: %v", res.Database, serr)
	}
	if !errors.Is(err, ErrTarget) {
		t.Errorf("err = %v, want ErrTarget (folder holds a foreign backup without manifest)", err)
	}
}

// C5b: Abbruch der ersten Sicherung nach dem Kopieren eines Audit-Archivs
// (vor dem Manifest): Der Rest enthält audit-archive-*.db; der nächste Lauf
// weist den Ordner weiter ab.
func TestZZC5bLeftoverWithAuditArchive(t *testing.T) {
	e := newEnv(t)
	dir := filepath.Join(t.TempDir(), "backup")
	for _, f := range []string{EncryptNoteName, snapshotName(e.now, 1), "audit-archive-2026-01-01_2026-03-31.db"} {
		if err := os.MkdirAll(dir, 0o700); err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(filepath.Join(dir, f), []byte("x"), 0o600); err != nil {
			t.Fatal(err)
		}
	}
	if err := os.MkdirAll(filepath.Join(dir, "blobs"), 0o700); err != nil {
		t.Fatal(err)
	}
	if _, _, err := e.r.prepareFolder(dir); err != nil {
		t.Errorf("leftovers of an interrupted first backup with a copied audit archive: %v", err)
	}
}
