package ops

import (
	"context"
	"encoding/json"
	"os"
	"path/filepath"
	"testing"
	"time"
)

// C1: Erster Export nach dem Update in denselben Ordner unter anderer
// Schreibweise (Altbestand-Manifest), Abbruch nach dem ersten Nachweis,
// dann der nächste Lauf (wieder über die neue Schreibweise).
func TestZZC1AbortAfterFirstRecordOtherSpelling(t *testing.T) {
	base := t.TempDir()
	dir := filepath.Join(base, "audit-export")
	now := time.Now().UTC()
	st := t3nOpen(t, filepath.Join(base, "d", "postkauz.db"))
	t3nEvent(t, st, now.Add(-72*time.Hour), "sync.run")
	t3nEvent(t, st, now.Add(-48*time.Hour), "sync.run")
	t3nEvent(t, st, now.Add(-24*time.Hour), "sync.run")
	t3nEvent(t, st, now.Add(-time.Minute), "sync.run")
	if _, err := t3nExport(st, dir); err != nil {
		t.Fatal(err)
	}
	t3n2LegacyManifest(t, dir)
	l := filepath.Join(base, "H-drive")
	if err := os.Symlink(base, l); err != nil {
		t.Skipf("no symlink: %v", err)
	}
	other := filepath.Join(l, "audit-export")
	t3nEvent(t, st, now, "keys.list")
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	repo := &t3n2CancelAfterSave{SQLite: st, cancel: cancel}
	if _, err := AuditExport(repo, other).Handler(ctx, Call{Input: json.RawMessage(`{}`)}); err == nil {
		t.Fatal("the export was not aborted")
	}
	if repo.n != 1 {
		t.Fatalf("records saved before the abort = %d, want 1", repo.n)
	}
	v1 := t3nVerify(t, st)
	t.Logf("verify right after the abort = %s %+v", v1.Status, v1.Problems)
	t3nEvent(t, st, now, "audit.export") // Abschluss-Eintrag des abgebrochenen Laufs
	if _, err := t3nExport(st, other); err != nil {
		t.Fatalf("next export (same spelling as the aborted run) = %v", err)
	}
	if v := t3nVerify(t, st); v.Status != "ok" {
		t.Errorf("verify after the next export = %+v", v)
	}
}
