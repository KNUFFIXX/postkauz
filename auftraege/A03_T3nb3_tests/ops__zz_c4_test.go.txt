package ops

import (
	"bytes"
	"os"
	"path/filepath"
	"testing"
	"time"
)

// C4: Eine innere Audit-Zeile eines abgeschlossenen Tages wird in der
// Datenbank verändert. Ein Export, der den Tag neu schreibt (anderer Ort
// bzw. Ort im Nachweis geleert), ersetzt die Datei im selben Ordner durch
// den veränderten Inhalt und überschreibt den Nachweis; verify meldet ok.
// Variante "other-spelling": Symlink auf denselben Ordner (Schreibrecht nur
// auf die DB für die Zeile). Variante "db-only": zusätzlich nur die DB
// (audit_exports.location = ”), Export mit unverändertem Pfad.
func TestZZC4RewriteReplacesEvidence(t *testing.T) {
	for _, tc := range []string{"other-spelling", "db-only"} {
		t.Run(tc, func(t *testing.T) {
			base := t.TempDir()
			dir := filepath.Join(base, "audit-export")
			now := time.Now().UTC()
			st := t3nOpen(t, filepath.Join(base, "d", "postkauz.db"))
			day1 := now.Add(-48 * time.Hour)
			t3nEvent(t, st, day1, "sync.run")
			t3nEvent(t, st, day1.Add(time.Second), "keys.list") // innere Zeile (id 2)
			t3nEvent(t, st, day1.Add(2*time.Second), "sync.run")
			t3nEvent(t, st, now.Add(-24*time.Hour), "sync.run")
			t3nEvent(t, st, now.Add(-time.Minute), "sync.run")
			if _, err := t3nExport(st, dir); err != nil {
				t.Fatal(err)
			}
			file := filepath.Join(dir, auditFileName(day1.Format("2006-01-02")))
			orig, _ := os.ReadFile(file)
			// Angreifer mit Schreibrecht auf die Datenbank: Sperr-Trigger weg.
			for _, q := range []string{`DROP TRIGGER audit_events_no_update`,
				`UPDATE audit_events SET operation = 'accounts.remove' WHERE id = 2`} {
				if err := st.ExecForTest(q); err != nil {
					t.Fatal(err)
				}
			}
			if v := t3nVerify(t, st); v.Status == "ok" {
				t.Fatalf("verify does not see the changed row: %+v", v)
			}
			target := dir
			switch tc {
			case "other-spelling":
				l := filepath.Join(base, "H-drive")
				if err := os.Symlink(base, l); err != nil {
					t.Skipf("no symlink: %v", err)
				}
				target = filepath.Join(l, "audit-export")
			case "db-only":
				if err := st.ExecForTest(`UPDATE audit_exports SET location = '' WHERE day = ?`, day1.Format("2006-01-02")); err != nil {
					t.Fatal(err)
				}
			}
			_, err := t3nExport(st, target)
			t.Logf("export = %v", err)
			after, _ := os.ReadFile(file)
			if !bytes.Equal(orig, after) {
				t.Errorf("the export replaced the original evidence of %s with content from the changed database", filepath.Base(file))
			}
			if v := t3nVerify(t, st); v.Status == "ok" {
				t.Errorf("verify after the export = ok; the changed row is no longer detectable")
			}
		})
	}
}
